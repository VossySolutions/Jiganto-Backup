function CascadeView({ layers, data, activeLayer, hoveredItem, setHoveredItem }) {
  return (
    <div style={{ display: "flex", flexDirection: "column", gap: 3 }}>
      {layers.map((layer, li) => {
        const items = data[layer.key];
        const isActive = !activeLayer || activeLayer === layer.key;
        const indent = li * 18;

        return (
          <div key={layer.key} style={{
            marginLeft: indent,
            opacity: isActive ? 1 : 0.35,
            transition: "all 0.3s",
          }}>
            {/* Layer Header */}
            <div style={{
              display: "flex", alignItems: "center", gap: 10,
              padding: "10px 16px 8px",
              marginBottom: 4,
            }}>
              <div style={{
                width: 28, height: 28, borderRadius: 8,
                background: layer.bg, border: `1px solid ${layer.border}`,
                display: "flex", alignItems: "center", justifyContent: "center",
                fontSize: 13, color: layer.accent,
              }}>{layer.icon}</div>
              <span style={{ fontSize: 11, fontWeight: 800, letterSpacing: "0.1em", textTransform: "uppercase", color: layer.accent }}>
                {layer.label}
              </span>
              <div style={{ flex: 1, height: 1, background: `linear-gradient(to right, ${layer.border}, transparent)` }} />
              <span style={{ fontSize: 11, color: "#334155" }}>{items.length} items</span>
            </div>

            {/* Items Row */}
            <div style={{ display: "flex", gap: 8, flexWrap: "wrap", paddingLeft: 8 }}>
              {items.map(item => (
                <div key={item.id}
                  onMouseEnter={() => setHoveredItem(item.id)}
                  onMouseLeave={() => setHoveredItem(null)}
                  style={{
                    background: hoveredItem === item.id ? layer.bg : "rgba(255,255,255,0.03)",
                    border: `1px solid ${hoveredItem === item.id ? layer.border : "rgba(255,255,255,0.06)"}`,
                    borderRadius: 10, padding: "10px 14px",
                    cursor: "pointer", transition: "all 0.2s",
                    transform: hoveredItem === item.id ? "translateY(-2px)" : "none",
                    maxWidth: 240, minWidth: 160,
                    position: "relative", overflow: "hidden",
                  }}>
                  <div style={{
                    position: "absolute", top: 0, left: 0, right: 0, height: 2,
                    background: `linear-gradient(to right, ${layer.accent}, transparent)`,
                  }} />
                  <div style={{ fontSize: 12, fontWeight: 600, color: "#cbd5e1", lineHeight: 1.4 }}>{item.name}</div>
                  {item.rag && (
                    <div style={{
                      marginTop: 8, display: "inline-flex", alignItems: "center", gap: 5,
                      background: `${ragColors[item.rag]}22`,
                      border: `1px solid ${ragColors[item.rag]}44`,
                      borderRadius: 5, padding: "2px 8px",
                    }}>
                      <div style={{ width: 6, height: 6, borderRadius: "50%", background: ragColors[item.rag] }} />
                      <span style={{ fontSize: 10, fontWeight: 700, color: ragColors[item.rag], letterSpacing: "0.05em" }}>{item.rag}</span>
                    </div>
                  )}
                </div>
              ))}
            </div>
          </div>
        );
      })}
    </div>
  );
}

function PyramidView({ layers, data }) {
  const reversed = [...layers].reverse();
  const maxWidth = 960;

  return (
    <div style={{ display: "flex", flexDirection: "column", alignItems: "center", gap: 3 }}>
      <div style={{ fontSize: 11, color: "#475569", letterSpacing: "0.08em", textTransform: "uppercase", marginBottom: 20 }}>
        Strategic Hierarchy — From Vision to Execution
      </div>
      {reversed.map((layer, li) => {
        const items = data[layer.key];
        const widthPct = 20 + (li / (reversed.length - 1)) * 80;
        const width = `${widthPct}%`;

        return (
          <div key={layer.key} style={{
            width,
            maxWidth,
            background: layer.bg,
            border: `1px solid ${layer.border}`,
            borderRadius: li === 0 ? "12px 12px 4px 4px" : li === reversed.length - 1 ? "4px 4px 12px 12px" : 4,
            padding: "14px 20px",
            transition: "all 0.3s",
            position: "relative",
            overflow: "hidden",
          }}>
            <div style={{
              position: "absolute", top: 0, left: 0, bottom: 0, width: 3,
              background: layer.accent,
            }} />
            <div style={{ display: "flex", alignItems: "center", justifyContent: "space-between" }}>
              <div style={{ display: "flex", alignItems: "center", gap: 10 }}>
                <span style={{ fontSize: 11, fontWeight: 800, letterSpacing: "0.08em", textTransform: "uppercase", color: layer.accent }}>
                  {layer.icon} {layer.label}
                </span>
                <span style={{ fontSize: 10, color: "#334155" }}>{items.length} items</span>
              </div>
              <div style={{ display: "flex", gap: 6, flexWrap: "wrap", justifyContent: "flex-end" }}>
                {items.slice(0, 3).map(item => (
                  <div key={item.id} style={{
                    background: "rgba(0,0,0,0.3)", border: "1px solid rgba(255,255,255,0.06)",
                    borderRadius: 6, padding: "4px 10px", fontSize: 11, color: "#94a3b8",
                  }}>{item.name.length > 28 ? item.name.slice(0, 28) + "…" : item.name}</div>
                ))}
                {items.length > 3 && (
                  <div style={{ fontSize: 11, color: "#475569", padding: "4px 8px" }}>+{items.length - 3} more</div>
                )}
              </div>
            </div>
          </div>
        );
      })}
    </div>
  );
}

function FlowView({ layers, data }) {
  return (
    <div style={{ overflowX: "auto", paddingBottom: 20 }}>
      <div style={{ display: "flex", gap: 0, minWidth: 900, alignItems: "stretch" }}>
        {layers.map((layer, li) => {
          const items = data[layer.key];
          const isLast = li === layers.length - 1;

          return (
            <div key={layer.key} style={{ display: "flex", alignItems: "center", flex: 1 }}>
              <div style={{
                flex: 1,
                background: layer.bg,
                border: `1px solid ${layer.border}`,

                <div style={{ display: "flex", flexDirection: "column", gap: 6 }}>
                  {items.map(item => (
